@startuml

participant "：蔵書管理" as Control
activate Control

participant "：蔵書リスト" as List

Control -> List  : かんたんUML入門書　＝　蔵書を取得する（蔵書番号）
activate List
deactivate List

participant "かんたんUML入門：蔵書" as UMLBook

Control -> UMLBook     :　　　　　　　　　　　貸出状況＝貸出中か調べる
activate UMLBook
deactivate UMLBook

group opt [貸し出し状況　＝　貸出無し]
    Control -> UMLBook    :　　　　　　　　　　　貸出状況を変更する（貸出中）
    activate UMLBook
    deactivate UMLBook
end
@enduml
